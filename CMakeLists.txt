set_property(GLOBAL PROPERTY JOB_POOLS raynmea_link=2)
set(CMAKE_JOB_POOL_LINK raynmea_link)
option(BUILD_TESTING "Configure isolated regression targets and CTest" OFF)
option(RAYNMEA_DEPLOYMENT "Build the approved bundled content for private deployment" OFF)
if(NOT WIN32 AND NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" AND NOT CMAKE_SYSTEM_NAME STREQUAL "Darwin")
    message(FATAL_ERROR "Supported native build platforms: Windows, Linux and macOS")
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "Darwin")
    if(NOT CMAKE_OSX_ARCHITECTURES STREQUAL "arm64")
        message(FATAL_ERROR "Initial macOS qualification requires explicit -DCMAKE_OSX_ARCHITECTURES=arm64")
    endif()
endif()
if(WIN32)
    enable_language(RC)
    if(NOT MINGW OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
        message(FATAL_ERROR "Select the accepted 64-bit MinGW Qt kit")
    endif()
endif()

# Explicitly compile the accepted forms. AUTOUIC otherwise prefers obsolete
# same-named forms beside some .cpp files over ui/widgets (for example Charger).
qt_wrap_ui(RAYNMEA_GENERATED_UI ${RAYNMEA_FORMS})
add_custom_target(raynmea_ui DEPENDS ${RAYNMEA_GENERATED_UI})

function(raynmea_common target)
    target_compile_definitions(${target} PUBLIC APP_VERSION="${PROJECT_VERSION}" APP_NAME="RayNmeaSimulator" APP_COMPANY="RFStateSide")
    target_include_directories(${target} PUBLIC "${PROJECT_SOURCE_DIR}/src" "${PROJECT_SOURCE_DIR}/src/widgets")
    target_link_libraries(${target} PUBLIC Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Network Qt6::SerialPort Qt6::Xml Qt6::Charts Qt6::Positioning Qt6::WebSockets raynmea_marine raynmea_sdl Threads::Threads)
    if(WIN32)
        target_link_libraries(${target} PUBLIC advapi32)
    elseif(CMAKE_SYSTEM_NAME STREQUAL "Darwin")
        target_link_libraries(${target} PUBLIC "-framework Security" "-framework CoreFoundation")
    endif()
    target_compile_definitions(${target} PRIVATE "$<$<CONFIG:Debug>:RAYNMEA_TEMPLATE_PROJECT_ROOT=\"${PROJECT_SOURCE_DIR}\">")
    set_target_properties(${target} PROPERTIES AUTOUIC_SEARCH_PATHS "${PROJECT_SOURCE_DIR}/ui;${PROJECT_SOURCE_DIR}/ui/widgets")
    set_target_properties(${target} PROPERTIES AUTOUIC OFF)
    target_include_directories(${target} BEFORE PUBLIC "${CMAKE_BINARY_DIR}")
    set_target_properties(${target} PROPERTIES AUTOGEN_BUILD_DIR "${CMAKE_CURRENT_BINARY_DIR}/${target}_moc")
    target_include_directories(${target} PUBLIC "${CMAKE_CURRENT_BINARY_DIR}/${target}_moc/include")
    add_dependencies(${target} raynmea_verify_dependencies)
    add_dependencies(${target} raynmea_ui)
endfunction()

function(raynmea_resources target isolated)
    add_library(${target} OBJECT EXCLUDE_FROM_ALL "${PROJECT_SOURCE_DIR}/RayNmeaSimulator.qrc" "${PROJECT_SOURCE_DIR}/resources/nmea2000/nmea2000.qrc" "${PROJECT_SOURCE_DIR}/resources/help/help.qrc")
    target_link_libraries(${target} PRIVATE Qt6::Core)
    raynmea_template_files(files ${isolated})
    qt_add_resources(${target} ${target}_templates PREFIX /templates BASE "${PROJECT_SOURCE_DIR}/resources/templates" FILES ${files})
    set(default_layout "${PROJECT_SOURCE_DIR}/resources/defaults/defaultLayout.json")
    # Existing unit scenarios retain their historical synthetic fixture. The
    # bundled-content/parity tests explicitly link the actual application resources.
    if("TEST_FIXTURE_DEFAULT" IN_LIST ARGN)
        set(default_layout "${PROJECT_SOURCE_DIR}/tests/fixtures/defaultLayout.json")
    endif()
    get_filename_component(default_directory "${default_layout}" DIRECTORY)
    qt_add_resources(${target} ${target}_default_layout PREFIX /defaults BASE "${default_directory}" FILES "${default_layout}")
endfunction()

    configure_file(cmake/RayNmeaSimulator.rc.in "${CMAKE_CURRENT_BINARY_DIR}/RayNmeaSimulator.rc" @ONLY)

    # Preserve the established system paths; DESTDIR provides private staging.
    # Runtime bundling and installer generation remain separate deployment work.
    install(TARGETS RayNmeaSimulator RUNTIME DESTINATION /opt/RayNmeaSimulator/bin)
    install(FILES packaging/com.rfstateside.RayNmeaSimulator.desktop DESTINATION /usr/share/applications)

    set(test_sources ${RAYNMEA_SOURCES})

    # acceptance fixture supplies its existing in-memory backend instead.
    add_library(raynmea_test_credentials STATIC EXCLUDE_FROM_ALL src/credentialstore.cpp)
    raynmea_common(raynmea_test_credentials)
    target_compile_definitions(raynmea_test_credentials PRIVATE RAYNMEA_TESTING)

        # Compile-time substitution: test executables cannot call real SecItem APIs.
        target_sources(raynmea_test_credentials PRIVATE src/credentialstore_macos.cpp tests/macos_credential_test_api.cpp)
    endif()
    add_library(raynmea_test_objects OBJECT EXCLUDE_FROM_ALL ${test_sources} ${RAYNMEA_HEADERS} ${RAYNMEA_FORMS})
    raynmea_common(raynmea_test_objects)
    target_compile_definitions(raynmea_test_objects PUBLIC RAYNMEA_TESTING)
    target_link_libraries(raynmea_test_objects PUBLIC Qt6::Test raynmea_test_credentials)
    raynmea_resources(raynmea_test_resources TRUE TEST_FIXTURE_DEFAULT)
    add_subdirectory(tests)
endif()
