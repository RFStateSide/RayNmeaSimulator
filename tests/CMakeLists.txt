# One target registration is the authoritative regression inventory.
function(ray_test name group)
    set(credential_options)
    if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND "CREDENTIALS" IN_LIST ARGN)
        list(APPEND credential_options --credentials)
    endif()
    set(resource_target raynmea_test_resources)
    if("APPLICATION_RESOURCES" IN_LIST ARGN)
        set(resource_target raynmea_app_resources)
    endif()
    qt_add_executable(${name} EXCLUDE_FROM_ALL "${PROJECT_SOURCE_DIR}/tests/${name}.cpp" $<TARGET_OBJECTS:${resource_target}>)
    target_link_libraries(${name} PRIVATE raynmea_test_objects)
    set_target_properties(${name} PROPERTIES AUTOUIC OFF WIN32_EXECUTABLE FALSE)
    target_compile_definitions(${name} PRIVATE QT_TESTCASE_SOURCEDIR="${PROJECT_SOURCE_DIR}/tests")
    add_test(NAME ${name} COMMAND "${Python3_EXECUTABLE}" "${PROJECT_SOURCE_DIR}/scripts/run-cmake-test.py" --name ${name} --root "${PROJECT_SOURCE_DIR}" --build "${CMAKE_BINARY_DIR}" --qt-root "${RAY_QT_ROOT}" --sdl-root "${SDL2_ROOT}" ${credential_options} -- $<TARGET_FILE:${name}>)
    set_tests_properties(${name} PROPERTIES LABELS "${group};synthetic" TIMEOUT 180 REQUIRED_FILES "$<TARGET_FILE:${name}>")
    if("CREDENTIALS" IN_LIST ARGN)
        set_property(TEST ${name} APPEND PROPERTY LABELS credentials)
        if(CMAKE_SYSTEM_NAME STREQUAL "Darwin")
            set_property(TEST ${name} APPEND PROPERTY LABELS "unit;credential-double")
        endif()
    endif()
    if(NOT TARGET ray_tests_${group})
        add_custom_target(ray_tests_${group})
    endif()
    add_dependencies(ray_tests_${group} ${name})
endfunction()
